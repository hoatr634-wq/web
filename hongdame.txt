    async function waitForElementToDisappear(element) {
        let maxWait = 50;
        while (maxWait > 0) {
            if (!element.isConnected || element.offsetParent === null) {
                return true;
            }
            await sleep(100);
            maxWait--;
        }
        return false;
    }

    async function waitForGlobalLoading(statusCallback) {
        let maxWait = 100;
        let isWaiting = false;

        while (maxWait > 0) {
            let loadingIcon = document.querySelector('[role="progressbar"], img[src*="loading"], i[class*="loading"], div[class*="spinner"]');
            
            if (loadingIcon && loadingIcon.offsetParent !== null) {
                isWaiting = true;
                if(statusCallback) statusCallback("⏳ Đang đợi loading...", "#FF69B4");
                await sleep(100);
                maxWait--;
            } else {
                if(isWaiting) await sleep(500); 
                return;
            }
        }
    }

    function findButtonFromV20_3(keywords) {
        let all = [...document.querySelectorAll('div[role="menuitem"], div[role="button"], button, span, div[role="dialog"] span, div[aria-label], div[tabindex="0"]')];
        all = all.filter(el => el.offsetParent !== null && el.offsetWidth > 0);
        for (let el of all) {
            let txt = (el.innerText || "").trim();
            let label = (el.getAttribute("aria-label") || "").trim();
            for (let k of keywords) {
                if (txt === k || label === k || txt.includes(k) || label.includes(k)) {
                    let parentBtn = el.closest('div[role="button"], button, div[role="menuitem"], div[tabindex="0"]');
                    return parentBtn ? parentBtn : el;
                }
            }
        }
        return null;
    }

    async function executeReport(reportConfig, statusCallback, progressCallback) {
        const steps = reportConfig.steps;

            if (shouldStop) return false;
            while (isPaused) { await sleep(500); if (shouldStop) return false; }

            if (progressCallback) progressCallback((stepNum / totalSteps) * 100);

            for(let retry=0; retry<10; retry++){
                if (shouldStop) break;
                
                if (step.xpath) {
                    el = getElementByXpath(step.xpath);
                }
                
                if (!el) {
                    if (step.name === "Next") {
                        el = findButtonFromV20_3(["Next", "Tiếp", "Tiếp tục"]);
                    } else if (step.name === "Submit") {
                        el = findButtonFromV20_3(["Submit", "Gửi", "Send"]);
                    } else if (step.name === "Done") {
                        el = findButtonFromV20_3(["Done", "Xong", "Hoàn tất", "Đóng", "Close"]);
                    } else if (step.name === "Send") {
                        el = findButtonFromV20_3(["Send", "Gửi", "Tiếp"]);
                    } else if (step.name === "Me") {
                        el = findButtonFromV20_3(["Me", "Tôi"]);
                        if(!el) el = getElementByXpath("//span[text()='Me']");
                        if(!el) el = getElementByXpath("//span[contains(text(),'Me')]");
                    } else if (step.name === "They're not a real person") {
                        el = findButtonFromV20_3(["They're not a real person", "Không phải người thật"]);
                        if(!el) el = getElementByXpath("//span[contains(text(),\"They're not a real person\")]");
                    } else if (step.name === "Bullying or harassment") {
                        el = findButtonFromV20_3(["Bullying or harassment", "Quấy rối"]);
                    } else {
                        el = findButtonFromV20_3([step.name]);
                    }
                }
                
                if(el) break;
                await sleep(800);
            }

            if (shouldStop) return false;

            if (el) {
                if (step.inputData) {
                    el.style.border = "2px solid #FF69B4";
                    simulateInput(el, step.inputData);
                    statusCallback("Chờ load list...", "#FF69B4");
                    await sleep(INPUT_DELAY);
                } else {
                    el.style.border = "2px solid #FF1493";
                    safeClick(el);
                    
                    if (step.name === "Next" || step.name === "Submit") {
                        statusCallback("Đợi nút ẩn...", "#FFB6C1");
                        await waitForElementToDisappear(el);
                    }
                    
                    await waitForGlobalLoading(statusCallback);
                    
                    statusCallback("Nghỉ 1s...", "#FF69B4");
                    await sleep(DELAY_TIME);
                }
            } else {
                console.error(`❌ LỖI: Không tìm thấy nút "${step.name}"`);
                statusCallback(`LỖI: Mất nút '${step.name}'`, "#FF1493");
            }
        }
        
        totalReportsDone++;
        return true;
    }

        if (btnResume) btnResume.onclick = () => { isPaused = false; updateUIState(); };
        if (btnStop) btnStop.onclick = () => { 
            if(confirm("🌸 DỪNG hoàn toàn?")) { 
                shouldStop = true; 
                isPaused = false; 
                isRunning = false;
                toggleAntiSleep(false);
                updateStatus("Đã dừng!", "#FF69B4");
            } 
        };
    }

    function updateUIState() {
        if(isMini) return;
        const btnStart = panel.querySelector("#btnStart");
        const btnPause = panel.querySelector("#btnPause");
        const btnStop = panel.querySelector("#btnStop");
        const btnResumeDiv = panel.querySelector("#btnResume");
        if (!btnStart) return;
        if (isRunning) { 
            btnStart.style.display = "none";
            btnStop.style.display = "block";
            if (isPaused) {
                btnPause.style.display = "none";
                btnResumeDiv.style.display = "block";
            } else {
                btnPause.style.display = "block";
                btnResumeDiv.style.display = "none";
            }
        } else { 
            btnStart.style.display = "block";
            btnPause.style.display = "none";
            btnStop.style.display = "none";
            btnResumeDiv.style.display = "none";
        }
    }

    function updateStatus(text, color = "#FFB6C1") {
        if (isMini) {
            const el = panel.querySelector("#miniStt");
            if (el) { el.innerText = text; el.style.color = color; }
        } else {
            const el = panel.querySelector("#stt");
            if (el) { el.innerText = text; el.style.color = color; }
        }
    }
    
    function updateProgress(percent) {
        if (isMini) {
            const el = panel.querySelector("#miniBar");
            if (el) el.style.width = percent + "%";
        } else {
            const el = panel.querySelector("#progressBar");
            if (el) el.style.width = percent + "%";
        }
    }

    function updateTotalDisplay() {
        if (isMini) {
            const el = panel.querySelector("#miniCount");
            if (el) el.innerText = `Đã báo cáo: ${totalReportsDone}`;
        } else {
            const el = panel.querySelector("#totalReports");
            if (el) el.innerText = totalReportsDone;
        }
    }

    function updateCurrentType(typeName) {
        if (!isMini && uiRefs.currentType) {
            uiRefs.currentType.innerText = typeName;
        }
    }

    // ==========================================
    // MAIN PROCESS (INFINITE LOOP)
    // ==========================================
    async function startProcess() {
        let useAntiSleep = true;
        if (!isMini) {
            useAntiSleep = panel.querySelector("#chkAntiSleep").checked;
        }

        shouldStop = false;
        isPaused = false;
        isRunning = true;
        currentReportIndex = 0;

        if (useAntiSleep) toggleAntiSleep(true);
        updateUIState();
        
        updateStatus("🚀 BẮT ĐẦU BÁO CÁO VÔ HẠN!", "#FF69B4");
        
        let loopCount = 0;

        while (!shouldStop) {
            loopCount++;
            
            for (let i = 0; i < reportTypes.length; i++) {
                if (shouldStop) break;
                
                currentReportIndex = i;
                const report = reportTypes[i];
                
                updateCurrentType(report.name);
                updateStatus(`📋 [Vòng ${loopCount}] Đang báo cáo: ${report.name}`, "#FFB6C1");
                updateTotalDisplay();
                
                const success = await executeReport(report, updateStatus, updateProgress);
                
                if (!success && !shouldStop) {
                    updateStatus(`⚠️ Lỗi khi báo cáo ${report.name}, tiếp tục...`, "#FF99CC");
                    await sleep(3000);
                }
                
                if (!shouldStop) {
                    await sleep(1500);
                }
            }
            
            if (!shouldStop) {
                updateStatus(`✅ Hoàn thành vòng ${loopCount}! Bắt đầu vòng mới...`, "#FF69B4");
                updateTotalDisplay();
                await sleep(2000);
            }
        }

        isRunning = false;
        toggleAntiSleep(false);
        updateUIState();
        
        if (!shouldStop) {
            updateStatus(`✨ ĐÃ DỪNG! Tổng số lần báo cáo: ${totalReportsDone}`, "#FF69B4");
            updateProgress(100);
        }
    }

    // Khởi tạo UI
    const initUI = () => {
        const refs = renderPanel();
        if (!isMini) {
            uiRefs = refs;
            updateTotalDisplay();
        }
        bindMainEvents();
    };
    
    initUI();
})();
